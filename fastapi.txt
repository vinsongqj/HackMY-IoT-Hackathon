class ComponentState(BaseModel):
    broken: bool
    isUnderMaintenance: bool
    maintenanceEtaSeconds: Optional[int] = None

class Barrier(ComponentState):
    name: str
    zoneParent: str
    state: Literal["Open", "Closed", "Opening", "Closing"]

class ParkingSpot(ComponentState):
    name: str
    purpose: Literal["Park", "EntrySpot", "ExitSpot"]
    parkingForCarType: Literal["Any", "Electric", "Accessible"]
    zoneParent: str
    detectedCars: list[str]

class Invoice(BaseModel):
    plate: str
    parkingCost: float
    chargingCost: float
    total: float
    issuedAt: str
    paidAt: Optional[str] = None

class Car(BaseModel):
    plate: str
    type: Literal["Electric", "Fuel"]
    currentSpot: Optional[str]
    parkedAt: Optional[str]
    parkingDurationMinutes: int
    invoice: Optional[Invoice]
    paymentStatus: Literal["None", "Requested", "Paid", "Validated", "Failed"]

class ExhaustFan(ComponentState):
    name: str
    zoneParent: str
    isOn: bool

class Light(BaseModel):
    name: str
    group: str
    zoneParent: str
    isOn: bool

class Zone(BaseModel):
    name: str
    gasCarbonMonoxideLevel: int
    risk: Literal["Safe", "Low", "Moderate", "High"]

class Penalty(BaseModel):
    id: str
    type: str
    component: str
    plate: Optional[str]
    fineAmount: float
    reason: str
    occurredAt: str

class OperatorSnapshot(BaseModel):
    barriers: list[Barrier]
    spots: list[ParkingSpot]
    cars: list[Car]
    fans: list[ExhaustFan]
    lights: list[Light]
    zones: list[Zone]
    penalties: list[Penalty]

class AdminStats(BaseModel):
    revenue: dict
    occupancy: dict
    penalties: dict

class RequestLog(BaseModel):
    id: str
    type: str
    timestamp: str
    plate: Optional[str]
    component: Optional[str]
    amount: Optional[float]
    metadata: dict

class LogPage(BaseModel):
    items: list[RequestLog]
    page: int
    pageSize: int
    total: int

# ------------------------------------------------------------------
# App lifecycle
# ------------------------------------------------------------------
state = {"client": None, "token": None}

@asynccontextmanager
async def lifespan(app: FastAPI):
    client = httpx.AsyncClient(base_url=SIMULATOR_URL, timeout=10.0)
    r = await client.post(
        "/api/v1/auth/login",
        json={"Email": SIMULATOR_EMAIL, "Password": SIMULATOR_PASSWORD},
    )
    r.raise_for_status()
    token = r.json()["token"]
    client.headers["Authorization"] = f"Bearer {token}"
    state["client"] = client
    state["token"] = token
    print(f"[backend] Logged in to simulator at {SIMULATOR_URL}")
    try:
        yield
    finally:
        await client.aclose()

app = FastAPI(lifespan=lifespan)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:5173"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# ------------------------------------------------------------------
# In-memory stores (replace with DB in production)
# ------------------------------------------------------------------
cars_db: dict[str, Car] = {}
penalties_db: list[Penalty] = []
logs_db: list[RequestLog] = []
subscribers: set[WebSocket] = set()

async def broadcast(payload: dict):
    dead = []
    for ws in list(subscribers):
        try:
            await ws.send_json(payload)
        except Exception:
            dead.append(ws)
    for ws in dead:
        subscribers.discard(ws)

def log_event(log_type: str, plate: Optional[str], component: Optional[str],
              amount: Optional[float], metadata: Optional[dict] = None):
    import uuid
    entry = RequestLog(
        id=str(uuid.uuid4()),
        type=log_type,
        timestamp=datetime.now(timezone.utc).isoformat(),
        plate=plate,
        component=component,
        amount=amount,
        metadata=metadata or {},
    )
    logs_db.append(entry)

@app.get("/api/operator/state", response_model=OperatorSnapshot)
async def operator_state():
    c: httpx.AsyncClient = state["client"]
    spots_r, barriers_r, zones_r = await asyncio.gather(
        c.get("/api/v1/list-parking-spots"),
        c.get("/api/v1/list-barriers"),
        c.get("/api/v1/list-zones"),
    )
    spots    = spots_r.json()
    barriers = barriers_r.json()
    zones    = zones_r.json()

    # Sync detected cars into cars_db
    for s in spots:
        for plate in s.get("detectedCars", []):
            if plate not in cars_db:
                cars_db[plate] = Car(
                    plate=plate,
                    type="Fuel",           # default; enrich from CarIn webhook
                    currentSpot=s["name"],
                    parkedAt=datetime.now(timezone.utc).isoformat(),
                    parkingDurationMinutes=0,
                    invoice=None,
                    paymentStatus="None",
                )

    return OperatorSnapshot(
        barriers=barriers,
        spots=spots,
        cars=list(cars_db.values()),
        fans=[],       # populate from /api/v1/list-exhaust-fans
        lights=[],     # populate from /api/v1/list-lights
        zones=zones,
        penalties=penalties_db[-200:],
    )

@app.post("/api/operator/cars/{plate}/invoice", response_model=Invoice)
async def issue_invoice(plate: str):
    car = cars_db.get(plate)
    if not car:
        raise HTTPException(404, "Car not found")
    if car.invoice:
        raise HTTPException(400, "Invoice already issued")

    parked = datetime.fromisoformat(car.parkedAt)
    minutes = int((datetime.now(timezone.utc) - parked).total_seconds() // 60)
    mult = ELECTRIC_MULTIPLIER if car.type == "Electric" else 1.0
    parking_cost  = minutes * PARKING_RATE_PER_MIN * mult
    charging_cost = minutes * PARKING_RATE_PER_MIN if car.type == "Electric" else 0.0

    invoice = Invoice(
        plate=plate,
        parkingCost=parking_cost,
        chargingCost=charging_cost,
        total=parking_cost + charging_cost,
        issuedAt=datetime.now(timezone.utc).isoformat(),
    )
    car.invoice = invoice
    car.parkingDurationMinutes = minutes
    car.paymentStatus = "Requested"
    return invoice

@app.get("/api/admin/stats", response_model=AdminStats)
async def admin_stats(date: Optional[str] = None):
    today = (date or datetime.now(timezone.utc).date().isoformat())

    paid_invoices = [c.invoice for c in cars_db.values() if c.invoice and c.paymentStatus == "Validated"]
    total_revenue = sum(i.total for i in paid_invoices)
    today_revenue = sum(i.total for i in paid_invoices if i.issuedAt.startswith(today))

    occupied = sum(1 for c in cars_db.values() if c.currentSpot)
    capacity = 20  # replace with real count from list-parking-spots

    return AdminStats(
        revenue={
            "total": total_revenue,
            "today": today_revenue,
            "thisWeek": total_revenue,
            "thisMonth": total_revenue,
        },
        occupancy={
            "current": occupied,
            "capacity": capacity,
            "today": len(cars_db),
            "peakToday": occupied,
        },
        penalties={
            "count": len(penalties_db),
            "totalAmount": sum(p.fineAmount for p in penalties_db),
            "byType": {},
        },
    )

@app.get("/api/admin/logs", response_model=LogPage)
async def admin_logs(
    type: Optional[str] = None,
    from_: Optional[str] = Query(None, alias="from"),
    to: Optional[str] = None,
    plate: Optional[str] = None,
    component: Optional[str] = None,
    page: int = 1,
    pageSize: int = 50,
):
    items = logs_db
    if type:      items = [l for l in items if l.type == type]
    if from_:     items = [l for l in items if l.timestamp >= from_]
    if to:        items = [l for l in items if l.timestamp <= to]
    if plate:     items = [l for l in items if l.plate == plate]
    if component: items = [l for l in items if l.component == component]

    total = len(items)
    start = (page - 1) * pageSize
    return LogPage(
        items=items[start:start + pageSize],
        page=page, pageSize=pageSize, total=total,
    )

@app.post("/webhook")
async def receive_webhook(payload: dict):
    print(f"[webhook] {payload}")
    etype = payload.get("type")

    if etype == "CarIn":
        plate = payload.get("plate")
        cars_db[plate] = Car(
            plate=plate,
            type="Electric" if payload.get("carType") == "Electric" else "Fuel",
            currentSpot=None,
            parkedAt=datetime.now(timezone.utc).isoformat(),
            parkingDurationMinutes=0,
            invoice=None,
            paymentStatus="None",
        )
        log_event("Arrival", plate, payload.get("spot"), None)

    elif etype == "CarOut":
        plate = payload.get("plate")
        log_event("Departure", plate, payload.get("spot"), None)
        cars_db.pop(plate, None)

    elif etype == "PaymentReceived":
        plate = payload.get("plate")
        car = cars_db.get(plate)
        if car and car.invoice:
            car.paymentStatus = "Paid"
            car.invoice.paidAt = datetime.now(timezone.utc).isoformat()
        log_event("Payment", plate, None, payload.get("amount"))

    elif etype == "Penalty":
        p = Penalty(**payload["penalty"])
        penalties_db.append(p)
        log_event("Penalty", p.plate, p.component, p.fineAmount, {"reason": p.reason})

    elif etype == "ComponentBroken":
        log_event("ComponentBroken", None, payload.get("name"), None,
                  {"problem": payload.get("problem")})

    await broadcast(payload)
    return {"ok": True}

@app.websocket("/ws/events")
async def events_ws(ws: WebSocket):
    await ws.accept()
    subscribers.add(ws)
    try:
        while True:
            await ws.receive_text()
    except WebSocketDisconnect:
        pass
    finally:
        subscribers.discard(ws)
